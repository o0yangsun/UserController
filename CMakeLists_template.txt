#${templateWarning}

set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_VERSION 1)

${cmakeRequiredVersion}

# ============================================================
# Cross compiler and tools
# ============================================================

set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_CXX_COMPILER arm-none-eabi-g++)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)

set(CMAKE_AR arm-none-eabi-ar)
set(CMAKE_OBJCOPY arm-none-eabi-objcopy)
set(CMAKE_OBJDUMP arm-none-eabi-objdump)
set(CMAKE_SIZE arm-none-eabi-size)

set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

# ============================================================
# Project settings
# ============================================================

project(${projectName} C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ============================================================
# MCU architecture
# ============================================================

set(MCPU "${mcpu}")

set(CPU_FLAGS
        -mcpu=${mcpu}
        -mthumb
)

# 根据 Cortex 内核选择浮点参数
if ("$${MCPU}" STREQUAL "cortex-m7")

    message(STATUS "Cortex-M7 hardware FPU enabled: FPv5-D16")

    set(FPU_FLAGS
            -mfpu=fpv5-d16
            -mfloat-abi=hard
    )

elseif ("$${MCPU}" STREQUAL "cortex-m4")

    message(STATUS "Cortex-M4 hardware FPU enabled: FPv4-SP-D16")

    set(FPU_FLAGS
            -mfpu=fpv4-sp-d16
            -mfloat-abi=hard
    )

else ()

    message(STATUS "Hardware FPU is not enabled for $${MCPU}")

    set(FPU_FLAGS
            -mfloat-abi=soft
    )

endif ()

# ============================================================
# Compiler options
# ============================================================

add_compile_options(
        $${CPU_FLAGS}
        $${FPU_FLAGS}

        -ffunction-sections
        -fdata-sections
        -fno-common
        -fmessage-length=0
)

# Enable assembler file preprocessing
add_compile_options(
        $<$<COMPILE_LANGUAGE:ASM>:-x$<SEMICOLON>assembler-with-cpp>
)

# Uncomment to suppress C++17 register keyword warnings
# set(CMAKE_CXX_FLAGS "$${CMAKE_CXX_FLAGS} -Wno-register")

# ============================================================
# Optimization settings
# ============================================================

if ("$${CMAKE_BUILD_TYPE}" STREQUAL "Release")

    message(STATUS "Maximum optimization for speed")
    add_compile_options(-Ofast)

elseif ("$${CMAKE_BUILD_TYPE}" STREQUAL "RelWithDebInfo")

    message(STATUS "Maximum optimization for speed with debug information")
    add_compile_options(-Ofast -g)

elseif ("$${CMAKE_BUILD_TYPE}" STREQUAL "MinSizeRel")

    message(STATUS "Maximum optimization for size")
    add_compile_options(-Os)

else ()

    message(STATUS "Minimal optimization with debug information")
    add_compile_options(-Og -g)

endif ()

# ============================================================
# Include directories and preprocessor definitions
# ============================================================

include_directories(
        ${includes}
)

add_definitions(
        ${defines}
)

# ============================================================
# Source files
# ============================================================

file(GLOB_RECURSE SOURCES
        ${sources}
)

# ============================================================
# Linker settings
# ============================================================

set(LINKER_SCRIPT
        $${CMAKE_SOURCE_DIR}/${linkerScript}
)

add_link_options(
        $${CPU_FLAGS}
        $${FPU_FLAGS}

        -T$${LINKER_SCRIPT}

        -Wl,-gc-sections
        -Wl,--print-memory-usage
        -Wl,-Map=$${PROJECT_BINARY_DIR}/$${PROJECT_NAME}.map
)

# ============================================================
# Executable target
# ============================================================

add_executable(
        $${PROJECT_NAME}.elf
        $${SOURCES}
        $${LINKER_SCRIPT}
)

# ============================================================
# HEX and BIN generation
# ============================================================

set(HEX_FILE
        $${PROJECT_BINARY_DIR}/$${PROJECT_NAME}.hex
)

set(BIN_FILE
        $${PROJECT_BINARY_DIR}/$${PROJECT_NAME}.bin
)

add_custom_command(
        TARGET $${PROJECT_NAME}.elf
        POST_BUILD

        COMMAND $${CMAKE_OBJCOPY}
                -Oihex
                $<TARGET_FILE:$${PROJECT_NAME}.elf>
                $${HEX_FILE}

        COMMAND $${CMAKE_OBJCOPY}
                -Obinary
                $<TARGET_FILE:$${PROJECT_NAME}.elf>
                $${BIN_FILE}

        COMMAND $${CMAKE_SIZE}
                $<TARGET_FILE:$${PROJECT_NAME}.elf>

        COMMENT
                "Building $${HEX_FILE}
Building $${BIN_FILE}"
)
